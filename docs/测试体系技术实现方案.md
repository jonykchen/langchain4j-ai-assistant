### 2.5 页面对象模型

```typescript
// frontend/tests/e2e/pages/ChatPage.ts
import { Page, Locator, expect } from '@playwright/test';

export class ChatPage {
  readonly page: Page;
  readonly messageInput: Locator;
  readonly sendButton: Locator;
  readonly messageList: Locator;
  readonly assistantMessage: Locator;
  readonly thinkingSection: Locator;
  readonly codeBlock: Locator;

  constructor(page: Page) {
    this.page = page;
    this.messageInput = page.locator('[data-testid="message-input"]');
    this.sendButton = page.locator('[data-testid="send-button"]');
    this.messageList = page.locator('[data-testid="message-list"]');
    this.assistantMessage = page.locator('[data-testid="assistant-message"]');
    this.thinkingSection = page.locator('[data-testid="thinking-section"]');
    this.codeBlock = page.locator('[data-testid="code-block"]');
  }

  async sendMessage(message: string) {
    await this.messageInput.fill(message);
    await this.sendButton.click();
  }

  async waitForResponse(timeout = 30000) {
    await this.assistantMessage.waitFor({ state: 'visible', timeout });
  }

  async expectMessageContains(text: string) {
    await expect(this.assistantMessage).toContainText(text);
  }

  async expandThinking() {
    await this.thinkingSection.click();
  }

  async copyCode() {
    await this.codeBlock.locator('[data-testid="copy-button"]').click();
  }
}
```

```typescript
    });

    await chatPage.sendMessage('讲个笑话');
    await chatPage.waitForResponse();

    // 验证消息逐步显示
    const messageText = await chatPage.assistantMessage.textContent();

    // Mock 包含思考过程的响应

      await route.fulfill({
        status: 200,
        body: JSON.stringify({
          code: 200,
          data: {
            reply: '<thinking>正在思考...</thinking>这是我的回答。'
          }
        })
      });
    });

    await chatPage.sendMessage('复杂问题');
    await chatPage.waitForResponse();

    // 验证思考部分存在且可展开
    await expect(chatPage.thinkingSection).toBeVisible();
    await chatPage.expandThinking();
    await expect(chatPage.thinkingSection).toContainText('正在思考');
  });

  test('代码块可以复制', async ({ page, context }) => {
    // 授予剪贴板权限
    await context.grantPermissions(['clipboard-read', 'clipboard-write']);

    await page.route('**/api/chat', async route => {
      await route.fulfill({
        body: JSON.stringify({
          code: 200,
          data: { reply: '```java\nSystem.out.println("Hello");\n```' }

      });

    await chatPage.sendMessage('写个Java示例');
    await chatPage.waitForResponse();

    expect(clipboardText).toContain('System.out.println');

  test('网络错误时显示友好提示', async ({ page }) => {
```

### 3.4 聊天场景模拟

```scala
// src/test/scala/gatling/simulations/ChatSimulation.scala
package gatling.simulations

import io.gatling.core.Predef._
import io.gatling.http.Predef._
import scala.concurrent.duration._
import scala.util.Random

class ChatSimulation extends Simulation {

  // HTTP 配置
  val httpProtocol = http
    .baseUrl("http://localhost:8082")
    .acceptHeader("application/json")
    .contentTypeHeader("application/json")
    .userAgentHeader("Gatling-Performance-Test")

  // 测试用户 Feeder
  val testUsers = List(
    ("user1", "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."),
    ("user2", "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...")
  )

  val userFeeder = testUsers.map { case (username, token) =>
    Map("username" -> username, "token" -> token)
  }.circular

  // 测试消息
  val messages = List(
    "你好",
    "今天天气怎么样？",
    "写一个冒泡排序算法",
    "解释一下 Spring Boot 的特性",
    "用 Python 实现一个简单的 HTTP 服务器"
  )

  val messageFeeder = Iterator.continually(Map("message" -> messages(Random.nextInt(messages.length))))

  // 聊天场景
  val chatScenario = scenario("Chat API Load Test")
    .feed(userFeeder)
    .feed(messageFeeder)
    .exec(
      http("发送聊天消息")
        .post("/api/chat")
        .header("Authorization", "Bearer ${token}")
        .body(StringBody("""{"message": "${message}"}""")).asJson
        .check(status.is(200))
        .check(jsonPath("$.code").is("200"))
        .check(jsonPath("$.data.reply").exists)
        .check(responseTimeInMillis.lt(5000)) // 响应时间 < 5s
    )
    .pause(1, 3) // 用户思考时间 1-3 秒

  // 流式聊天场景
  val streamScenario = scenario("Stream API Load Test")
    .feed(userFeeder)
    .feed(messageFeeder)
    .exec(
      http("发送流式消息")
        .post("/api/chat/stream")
        .header("Authorization", "Bearer ${token}")
        .header("Accept", "text/event-stream")
        .body(StringBody("""{"message": "${message}"}""")).asJson
        .check(status.is(200))
        .check(header("Content-Type").is("text/event-stream;charset=UTF-8"))
    )
    .pause(2, 5)

  // 健康检查场景
  val healthScenario = scenario("Health Check")
    .exec(
      http("健康检查")
        .get("/actuator/health")
        .check(status.is(200))
    )

  // 负载配置
  val standardLoad = Seq(
    rampUsers(10).during(10.seconds),    // 10 秒内启动 10 用户
    constantUsersPerSec(5).during(1.minute), // 持续 5 用户/秒 1 分钟
    rampUsersPerSec(5).to(20).during(2.minutes), // 5->20 用户/秒
    constantUsersPerSec(20).during(2.minute),  // 持续 20 用户/秒 2 分钟
    rampUsersPerSec(20).to(5).during(30.seconds)  // 降温
  )

  // 压力测试配置
  val stressLoad = Seq(
    rampUsersPerSec(10).to(100).during(5.minutes),
    constantUsersPerSec(100).during(3.minutes),
    rampUsersPerSec(100).to(10).during(1.minute)
  )

  // 执行配置
  setUp(
    chatScenario.inject(standardLoad)
      .protocols(httpProtocol)
      .assertions(
        global.responseTime.max.lt(10000),
        global.responseTime.mean.lt(3000),
        global.successfulRequests.percent.gt(95),
        global.responseTime.percentile3.lt(5000)
      ),
    streamScenario.inject(rampUsers(5).during(30.seconds))
      .protocols(httpProtocol)
      .andThen(
        healthScenario.inject(atOnceUsers(1))
          .protocols(httpProtocol)
      )
  )
}
```

```java
// src/test/java/com/jonychen/ai/evaluator/ResponseQualityEvaluator.java
package com.jonychen.ai.evaluator;

import com.jonychen.ai.model.AIModelTestCase;
import com.jonychen.ai.model.AIModelTestResult;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.regex.Pattern;

/**
 * 响应质量评估器
 */
@Component
public class ResponseQualityEvaluator {

    /**
     * 评估响应质量
     */
    public AIModelTestResult.AssertionResult evaluate(
            AIModelTestCase testCase,
            String actualResponse,
            long responseTimeMs
    ) {
        List<String> passed = new ArrayList<>();
        List<String> failed = new ArrayList<>();

        for (var assertion : testCase.assertions()) {
            boolean result = switch (assertion.type()) {
                case "CONTAINS" -> {
                    boolean passed_check = actualResponse.contains(assertion.expected().toString());
                    yield evaluateAndRecord(passed_check, assertion, passed, failed);
                }
                case "NOT_CONTAINS" -> {
                    boolean passed_check = !actualResponse.contains(assertion.expected().toString());
                    yield evaluateAndRecord(passed_check, assertion, passed, failed);
                }
                case "MATCHES_REGEX" -> {
                    Pattern pattern = Pattern.compile(assertion.expected().toString());
                    yield evaluateAndRecord(pattern.matcher(actualResponse).find(), assertion, passed, failed);
                }
                case "RESPONSE_TIME_MS" -> {
                    long maxTime = Long.parseLong(assertion.expected().toString());
                    yield evaluateAndRecord(responseTimeMs <= maxTime, assertion, passed, failed);
                }
                default -> evaluateAndRecord(false, assertion, passed, failed);
            };
        }

        return new AIModelTestResult.AssertionResult(
            passed,
            failed,
            calculateScore(passed.size(), passed.size() + failed.size())
        );
    }

    private boolean evaluateAndRecord(boolean passed_check, AIModelTestCase.Assertion assertion,
                                       List<String> passed, List<String> failed) {
        if (passed_check) {
            passed.add(assertion.description());
        } else {
            failed.add(assertion.description() + " (expected: " + assertion.expected() + ")");
        }
        return passed_check;
    }

    private double calculateScore(int passed, int total) {
        return total == 0 ? 1.0 : (double) passed / total;
    }
}
```

### 5.4 ReAct Agent 测试

```java
// src/test/java/com/jonychen/ai/evaluator/ReActEvaluator.java
package com.jonychen.ai.evaluator;

import com.jonychen.planning.agent.ReActResult;
import com.jonychen.planning.agent.ReActStep;
import org.springframework.stereotype.Component;

import java.util.List;

/**
 * ReAct Agent 评估器
 */
@Component
public class ReActEvaluator {

    /**
     * 评估 ReAct 执行结果
     */
    public ReActEvaluationResult evaluate(ReActResult result, ReActTestCase testCase) {
        return ReActEvaluationResult.builder()
            .success(result.success())
            .iterations(result.iterations())
            .iterationEfficiency(calculateIterationEfficiency(result, testCase))
            .toolUsageCorrect(verifyToolUsage(result.steps(), testCase.expectedTools()))
            .thoughtQuality(evaluateThoughtQuality(result.steps()))
            .finalAnswerQuality(evaluateFinalAnswer(result, testCase))
            .build();
    }

    private double calculateIterationEfficiency(ReActResult result, ReActTestCase testCase) {
        int expectedIterations = testCase.expectedIterations();
        int actualIterations = result.iterations();

        if (actualIterations <= expectedIterations) {
            return 1.0;
        }
        // 超过预期迭代次数，扣分
        return Math.max(0, 1.0 - (actualIterations - expectedIterations) * 0.1);
    }

    private boolean verifyToolUsage(List<ReActStep> steps, List<String> expectedTools) {
        List<String> actualTools = steps.stream()
            .filter(ReActStep::hasAction)
            .map(ReActStep::action)
            .toList();

        // 检查是否调用了预期的工具
        for (String expectedTool : expectedTools) {
            if (!actualTools.contains(expectedTool)) {
                return false;
            }
        }
        return true;
    }

    private double evaluateThoughtQuality(List<ReActStep> steps) {
        // 评估思考过程质量
        return steps.stream()
            .filter(s -> s.thought() != null)
            .mapToDouble(s -> {
                String thought = s.thought();
                // 检查思考是否有意义
                if (thought.length() < 10) return 0.5;
                if (thought.contains("因为") || thought.contains("所以")) return 1.0;
                if (thought.contains("首先") || thought.contains("然后")) return 0.9;
                return 0.7;
            })
            .average()
            .orElse(0.5);
    }

    private double evaluateFinalAnswer(ReActResult result, ReActTestCase testCase) {
        if (!result.success()) return 0.0;
        if (result.finalAnswer() == null) return 0.0;

        String answer = result.finalAnswer();
        String expected = testCase.expectedAnswer();

        // 简单的关键词匹配
        long matchedKeywords = testCase.keywords().stream()
            .filter(answer::contains)
            .count();

        return (double) matchedKeywords / testCase.keywords().size();
    }

    public record ReActTestCase(
        String question,
        List<String> expectedTools,
        int expectedIterations,
        String expectedAnswer,
        List<String> keywords
    ) {}

    @lombok.Builder
    @lombok.Data
    public static class ReActEvaluationResult {
        private boolean success;
        private int iterations;
        private double iterationEfficiency;
        private boolean toolUsageCorrect;
        private double thoughtQuality;
        private double finalAnswerQuality;

        public double overallScore() {
            return (iterationEfficiency + (toolUsageCorrect ? 1.0 : 0.0) + thoughtQuality + finalAnswerQuality) / 4.0;
        }
    }
}
```

                ),
                null
            ),

                "tool-calling",

    @ParameterizedTest

        System.out.println("---");

### 7.2 TestExecutionService

```java
// src/main/java/com/jonychen/admin/service/TestExecutionService.java
package com.jonychen.admin.service;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;

@Slf4j
@Service
@RequiredArgsConstructor
public class TestExecutionService {

    private final Map<String, Process> runningProcesses = new ConcurrentHashMap<>();
    private final Map<String, TestJobStatus> jobStatuses = new ConcurrentHashMap<>();
    private final AIModelTestRunner aiModelTestRunner;

    public String runE2ETests() {
        String jobId = UUID.randomUUID().toString();
        new Thread(() -> {
            try {
                jobStatuses.put(jobId, new TestJobStatus(jobId, "running", System.currentTimeMillis()));

                ProcessBuilder pb = new ProcessBuilder(
                    "npx", "playwright", "test"
                );
                pb.directory(new File("frontend"));
                pb.redirectErrorStream(true);
                Process process = pb.start();
                runningProcesses.put(jobId, process);

                // 读取输出日志（仅记录，不用于解析结果）
                try (BufferedReader reader = new BufferedReader(
                        new InputStreamReader(process.getInputStream()))) {
                    String line;
                    while ((line = reader.readLine()) != null) {
                        // 剥离 ANSI 转义码后记录日志
                        String cleanLine = stripAnsiCodes(line);
                        log.info("[E2E] {}", cleanLine);
                    }
                }

                int exitCode = process.waitFor();

                // 从 JSON 报告文件解析结果（替代日志行解析）
                Path jsonReportPath = new File("frontend/test-results/report.json").toPath();
                List<TestResultSummary> results = parseJsonReport(jsonReportPath);
                jobStatuses.put(jobId, new TestJobStatus(
                    jobId,
                    exitCode == 0 ? "completed" : "failed",
                    jobStatuses.get(jobId).startTime(),
                    System.currentTimeMillis()
                ));
                runningProcesses.remove(jobId);
            } catch (Exception e) {
                log.error("E2E test execution failed", e);
                jobStatuses.put(jobId, new TestJobStatus(jobId, "failed",
                    jobStatuses.get(jobId).startTime(), System.currentTimeMillis()));
            }
        }).start();

        return jobId;
    }

    public String runPerformanceTest(String simulation) {
        String jobId = UUID.randomUUID().toString();
        new Thread(() -> {
            try {
                jobStatuses.put(jobId, new TestJobStatus(jobId, "running", System.currentTimeMillis()));

                ProcessBuilder pb = new ProcessBuilder(
                    "mvn", "gatling:test",
                    "-Dgatling.simulationClass=gatling.simulations." + simulation
                );
                Process process = pb.start();
                runningProcesses.put(jobId, process);

                int exitCode = process.waitFor();
                jobStatuses.put(jobId, new TestJobStatus(
                    jobId,
                    exitCode == 0 ? "completed" : "failed",
                    jobStatuses.get(jobId).startTime(),
                    System.currentTimeMillis()
                ));
                runningProcesses.remove(jobId);
            } catch (Exception e) {
                log.error("Performance test execution failed", e);
            }
        }).start();

        return jobId;
    }

    public String runAIModelTests(String category) {
        String jobId = UUID.randomUUID().toString();
        new Thread(() -> {
            try {
                jobStatuses.put(jobId, new TestJobStatus(jobId, "running", System.currentTimeMillis()));

                List<AIModelTestSummary> results = aiModelTestRunner.runTests(category);
                cacheAIResults(jobId, results);

                jobStatuses.put(jobId, new TestJobStatus(
                    jobId, "completed",
                    jobStatuses.get(jobId).startTime(),
                    System.currentTimeMillis()
                ));
            } catch (Exception e) {
                log.error("AI model test execution failed", e);
                jobStatuses.put(jobId, new TestJobStatus(jobId, "failed",
                    jobStatuses.get(jobId).startTime(), System.currentTimeMillis()));
            }
        }).start();

        return jobId;
    }

    public TestJobStatus getJobStatus(String jobId) {
        return jobStatuses.getOrDefault(jobId, new TestJobStatus(jobId, "not_found", 0, 0));
    }

    public void cancelJob(String jobId) {
        Process process = runningProcesses.get(jobId);
        if (process != null) {
            process.destroy();
            runningProcesses.remove(jobId);
            jobStatuses.put(jobId, new TestJobStatus(jobId, "cancelled",
                jobStatuses.get(jobId).startTime(), System.currentTimeMillis()));
        }
    }

    // ... 其他方法实现
}
```
